#### 3.2.2 创建API接口

```typescript
// src/api/custom.ts
import request from './request'
import type { Result, PageResult } from '@/types/common'

export interface CustomItem {
  id: number
  name: string
  createTime: string
}

export interface CustomQuery {
  keyword?: string
  page?: number
  pageSize?: number
}

/**
 * 查询列表
 */
export const getCustomList = (params: CustomQuery) => {
  return request<Result<PageResult<CustomItem>>>({
    url: '/v1/custom',
    method: 'get',
    params
  })
}

/**
 * 获取详情
 */
export const getCustomDetail = (id: number) => {
  return request<Result<CustomItem>>({
    url: `/v1/custom/${id}`,
    method: 'get'
  })
}

/**
 * 创建
 */
export const createCustom = (data: any) => {
  return request<Result<number>>({
    url: '/v1/custom',
    method: 'post',
    data
  })
}

/**
 * 删除
 */
export const deleteCustom = (id: number) => {
  return request<Result<void>>({
    url: `/v1/custom/${id}`,
    method: 'delete'
  })
}
```

#### 3.3.2 使用组合式函数

```typescript
// src/hooks/useTable.ts
import { ref, reactive } from 'vue'

export interface PaginationConfig {
  page: number
  pageSize: number
  total: number
}

export function useTable<T = any>(fetchApi: Function) {
  const loading = ref(false)
  const tableData = ref<T[]>([])

  const pagination = reactive<PaginationConfig>({
    page: 1,
    pageSize: 10,
    total: 0
  })

  const fetchList = async (params: Record<string, any> = {}) => {
    loading.value = true
    try {
      const { data } = await fetchApi({
        ...params,
        page: pagination.page,
        pageSize: pagination.pageSize
      })
      tableData.value = data.records
      pagination.total = data.total
    } finally {
      loading.value = false
    }
  }

  const handlePageChange = () => {
    fetchList()
  }

  return {
    loading,
    tableData,
    pagination,
    fetchList,
    handlePageChange
  }
}
```

使用示例:

```vue
<script setup lang="ts">
import { useTable } from '@/hooks/useTable'
import { getCustomList } from '@/api/custom'

  fetchList()
})
</script>
```

## 7. 常见问题排查

### 7.1 后端问题

**问题**: 启动报错 "Table 'xxx' doesn't exist"
**解决**: 检查数据库表是否创建,执行初始化SQL脚本

**问题**: Redis连接失败
**解决**: 检查Redis服务是否启动,配置是否正确

**问题**: 规则编译失败
**解决**: 检查DRL语法,查看详细错误日志

### 7.2 前端问题

**问题**: npm install失败
**解决**: 清除缓存 `npm cache clean --force`,删除node_modules重新安装

**问题**: 跨域问题
**解决**: 检查vite.config.ts的proxy配置,确保后端允许跨域

**问题**: 页面空白
**解决**: 检查控制台错误,查看路由配置是否正确
